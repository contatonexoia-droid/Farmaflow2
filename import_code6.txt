            if (!ean || ean.trim() === '') {

                    status: 'no-change', 
                    details: "Pulado: Código de barras (EAN) ausente" 
                });
                continue;
            }
            
            let existingProduct: Product | undefined = undefined;
            let isNewProductUpdate = false;
            
            if (ean) {
                existingProduct = productsByEan.get(String(ean));
                if (!existingProduct) {
                    existingProduct = newProductsByEan.get(String(ean));

            }
